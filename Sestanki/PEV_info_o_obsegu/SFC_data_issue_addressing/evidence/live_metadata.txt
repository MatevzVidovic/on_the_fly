public|pev_deli_stavb|call_sync_from_deli|O|CREATE TRIGGER call_sync_from_deli AFTER INSERT OR UPDATE OF id_rel_sta ON public.pev_deli_stavb FOR EACH ROW EXECUTE FUNCTION trg_call_sync_from_deli()|public|trg_call_sync_from_deli|CREATE OR REPLACE FUNCTION public.trg_call_sync_from_deli()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
declare
    v_old_pe uuid;
    v_new_pe uuid;
begin
    if TG_OP in ('INSERT', 'UPDATE') then
        if NEW.id_rel_sta is not null then
            select f.id_rel_pe
            into v_new_pe
            from pev_stavbe s
                     join pev_fo f on f.id = s.id_rel_fo
            where s.id = NEW.id_rel_sta;
            if v_new_pe is not null then perform sync_pe_usage(v_new_pe); end if;
        end if;
    end if;

    if TG_OP = 'UPDATE' then
        if OLD.id_rel_sta is not null then
            select f.id_rel_pe
            into v_old_pe
            from pev_stavbe s
                     join pev_fo f on f.id = s.id_rel_fo
            where s.id = OLD.id_rel_sta;
            if v_old_pe is not null and v_old_pe is distinct from v_new_pe then perform sync_pe_usage(v_old_pe); end if;
        end if;
    end if;

    return null;
end

public|pev_parcele|tr_pev_parcele_delez_bzps|O|CREATE TRIGGER tr_pev_parcele_delez_bzps BEFORE INSERT OR UPDATE OF delez ON public.pev_parcele FOR EACH ROW EXECUTE FUNCTION trg_pev_parcele_delez_bzps()|public|trg_pev_parcele_delez_bzps|CREATE OR REPLACE FUNCTION public.trg_pev_parcele_delez_bzps()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
declare
    v_id_rel_pe    uuid;
    v_valid_to     date;
    v_povrsina_zps numeric;
begin
    -- If this row is not flagged for geometry use, we skip.
    if not NEW.fo_geom_upost then return NEW; end if;

    -- Figure out which PE is attached
    select f.id_rel_pe into v_id_rel_pe from pev_fo f where f.id = NEW.id_rel_fo;

    -- If there's no PE, we null out delez_bzps and return.
    if v_id_rel_pe is null then
        NEW.delez_bzps := null;
        return NEW;
    end if;

    -- Grab 'valid_to' from the PE record, defaulting to 2100-01-01
    select coalesce(pe.valid_to, date '2100-01-01') into v_valid_to from pev_posebne_enote pe where pe.id = v_id_rel_pe;

    -- Summation of povrsina_zps for matching stavbe
    select sum(sub_kn.povrsina_zps)
    into v_povrsina_zps
    from pev_stavbe st
             join ( select distinct on (stavba_parcela_id, datum_od, datum_do) stavba_parcela_id,
                                                                               datum_do,
                                                                               datum_od,
                                                                               parcela_id,
                                                                               stavba_id,
                                                                               povrsina_zps
                    from kn_nep_stavbe_parcele_h
                    order by stavba_parcela_id, datum_od, datum_do, datum_sys desc ) as sub_kn
                  on sub_kn.parcela_id = NEW.parcela_id and v_valid_to <= sub_kn.datum_do and
                     st.stavba_id = sub_kn.stavba_id
    where st.id_rel_fo = NEW.id_rel_fo;

    -- If there's no match at all, sum(...) is NULL.
    if v_povrsina_zps is null then
        /*
          We have two common use-cases here:
            1) On INSERT, there's no “old” value, so you probably want to null out.
            2) On UPDATE, you might prefer to keep whatever was there.
        */
        if TG_OP = 'UPDATE' then
            -- Keep the old value
            NEW.delez_bzps := OLD.delez_bzps;
        else
            -- For INSERT or other operations, just null it out
            NEW.delez_bzps := null;
        end if;

    else
        -- If we do have a sum(...) => apply the difference
        if NEW.delez is not null then
            NEW.delez_bzps := greatest(NEW.delez - v_povrsina_zps, 0);
        else
            NEW.delez_bzps := null;
        end if;
    end if;

    return NEW;
end;
